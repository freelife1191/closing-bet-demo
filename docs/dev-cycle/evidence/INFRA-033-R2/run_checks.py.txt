def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def filtered_env() -> tuple[dict[str, str], list[str]]:
    env = os.environ.copy()
    removed = [key for key in env if key in {"OMX_ROOT", "OMX_STATE_ROOT", "OMX_SESSION_ID"} or key.startswith("OMX_TEAM_")]
    for key in removed:
        env.pop(key, None)
    return env, sorted(removed)


def extract_ids(value: Any) -> set[str]:
    """loaded/list의 data 배열과 알려진 thread ID 필드만 읽는다."""
    if isinstance(value, str):
        return {value}
    if isinstance(value, list):
        return set().union(*(extract_ids(item) for item in value)) if value else set()
    if isinstance(value, dict):
        found = {value["threadId"]} if isinstance(value.get("threadId"), str) else set()
        if isinstance(value.get("id"), str) and any(key in value for key in ("parentThreadId", "agentRole", "ephemeral")):
            found.add(value["id"])
        for key in ("data", "threads"):
            if isinstance(value.get(key), list):
                found.update(extract_ids(value[key]))
        return found
    return set()


def root_child_completion(events: list[dict[str, Any]], root_id: str, child_id: str) -> dict[str, bool]:
    result = {"root_final": False, "child_final": False, "root_completed": False, "child_completed": False}
    for event in events:
        params = event.get("params", {})
        thread_id = params.get("threadId")
        item = params.get("item", {})
        if event.get("method") == "item/completed" and item.get("type") == "agentMessage" and item.get("phase") == "final_answer":
            if thread_id == root_id:
                result["root_final"] = True
            elif thread_id == child_id:
                result["child_final"] = True
        if event.get("method") == "turn/completed" and params.get("turn", {}).get("status") == "completed":
            if thread_id == root_id:
                result["root_completed"] = True
            elif thread_id == child_id:
                result["child_completed"] = True
    return result


def local_checks() -> dict[str, Any]:
    seed = json.loads(SEED.read_text())
    assets = {path: sha(REPO / path) for path in ASSET_PATHS}
    legacy_assets = {path: digest for digest, path in (line.split(maxsplit=1) for line in seed["WORKFLOW_ASSET_SHA256.txt"].splitlines())}
    assert set(assets) == set(legacy_assets), "legacy seed asset path set drift"
    assert all(len(digest) == 64 for digest in assets.values()), "current asset SHA capture failed"
    assert extract_ids({"data": ["root", {"id": "child", "ephemeral": True}], "title": "not-an-id", "source": "vscode"}) == {"root", "child"}
    events = [
        {"method": "item/completed", "params": {"threadId": "child", "item": {"type": "agentMessage", "phase": "final_answer"}}},
        {"method": "turn/completed", "params": {"threadId": "child", "turn": {"status": "completed"}}},
        {"method": "item/completed", "params": {"threadId": "root", "item": {"type": "agentMessage", "phase": "final_answer"}}},
        {"method": "turn/completed", "params": {"threadId": "root", "turn": {"status": "completed"}}},
    ]
    assert all(root_child_completion(events, "root", "child").values())
    unrelated = [{"method": "item/completed", "params": {"threadId": "other", "item": {"type": "agentMessage", "phase": "final_answer"}}}, {"method": "turn/completed", "params": {"threadId": "other", "turn": {"status": "completed"}}}]
    assert not any(root_child_completion(unrelated, "root", "child").values())
    env, removed = filtered_env()
    assert "HOME" in env and "CODEX_HOME" not in removed
    return {"fixture_seed": str(SEED), "assets": assets, "actor": ACTOR, "removed_omx_keys": removed, "legacy_prefix_lines": 312, "model_calls_started": False}


def rpc_prepare() -> dict[str, Any]:
    assert CODEX.is_file(), f"missing app-server: {CODEX}"
    schema = subprocess.run([str(CODEX), "app-server", "generate-json-schema", "--out", str(ROOT / "schema")], text=True, capture_output=True, check=False)
    assert schema.returncode == 0, schema.stderr
    turn_schema = (ROOT / "schema/v2/TurnStartParams.json").read_text(encoding="utf-8")
    assert '"skill"' in turn_schema and '"SkillUserInput"' in turn_schema
    return {"app_server": str(CODEX), "actor": ACTOR, "trust_config": "projects.<exact fixture cwd>.trust_level=trusted (thread-local)", "skill_user_input_validated": True, "model_calls_started": False}


class Rpc:
    """실행 시에만 app-server를 시작하는 작은 JSON-RPC 드라이버."""
    def __init__(self, cwd: Path, artifact: Path) -> None:
        self.cwd, self.artifact, self.seq = cwd, artifact, 0
        self.events: queue.Queue[dict[str, Any]] = queue.Queue()
        self.raw: list[dict[str, Any]] = []
        self.proc: subprocess.Popen[str] | None = None

    def start(self) -> None:
        env, removed = filtered_env()
        self.artifact.mkdir(parents=True, exist_ok=True)
        err = (self.artifact / "app-server.stderr.log").open("w", encoding="utf-8")
        self.proc = subprocess.Popen([str(CODEX), "app-server", "--stdio"], cwd=self.cwd, env=env, stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=err, text=True, bufsize=1)
        def reader() -> None:
            assert self.proc and self.proc.stdout
            for line in self.proc.stdout:
                try:
                    msg = json.loads(line)
                except json.JSONDecodeError:
                    self.raw.append({"non_json": line})
                    continue
                self.raw.append(msg); self.events.put(msg)
        threading.Thread(target=reader, daemon=True).start()
        self.call("initialize", {"clientInfo": {"name": "infra-033-r2", "version": "1"}, "capabilities": {"experimentalApi": True}})
        self.send({"jsonrpc": "2.0", "method": "initialized", "params": {}})
        self.raw.append({"removed_omx_keys": removed, "actor": ACTOR})

    def send(self, msg: dict[str, Any]) -> None:
        assert self.proc and self.proc.stdin
        self.proc.stdin.write(json.dumps(msg, ensure_ascii=False) + "\n"); self.proc.stdin.flush()

    def call(self, method: str, params: dict[str, Any], timeout: int = 30) -> dict[str, Any]:
        self.seq += 1; ident = self.seq
        self.send({"jsonrpc": "2.0", "id": ident, "method": method, "params": params})
        until = time.monotonic() + timeout
        while time.monotonic() < until:
            msg = self.events.get(timeout=max(0.1, until - time.monotonic()))
            if msg.get("id") == ident:
                if "error" in msg: raise RuntimeError(str(msg["error"]))
                return msg.get("result", {})
        raise TimeoutError(method)

    def thread(self, sandbox: str) -> str:
        out = self.call("thread/start", {"cwd": str(self.cwd), "ephemeral": True, "sandbox": sandbox, "approvalPolicy": "never", "model": ACTOR["model"], "config": {"projects": {str(self.cwd): {"trust_level": "trusted"}}}})
        return out.get("thread", {}).get("id") or out["threadId"]

    def turn(self, thread_id: str, text: str, skill: bool = True, timeout: int = 1200) -> list[dict[str, Any]]:
        inputs: list[dict[str, Any]] = []
        if skill: inputs.append({"type": "skill", "name": "dev-cycle", "path": str(self.cwd / ".agents/skills/dev-cycle/SKILL.md")})
        inputs.append({"type": "text", "text": text, "text_elements": []})
        out = self.call("turn/start", {"threadId": thread_id, "input": inputs, "approvalPolicy": "never", "effort": ACTOR["effort"]})
        turn_id = out.get("turn", {}).get("id") or out.get("turnId")
        seen: list[dict[str, Any]] = []
        until = time.monotonic() + timeout
        while time.monotonic() < until:
            msg = self.events.get(timeout=max(0.1, until - time.monotonic()))
            seen.append(msg)
            if msg.get("method") == "turn/completed" and (msg.get("params", {}).get("turn", {}).get("id") in {None, turn_id}): return seen
        raise TimeoutError(f"turn:{turn_id}")

    def close(self) -> None:
        if self.proc and self.proc.poll() is None:
            self.proc.terminate()
            try: self.proc.wait(timeout=10)
            except subprocess.TimeoutExpired: self.proc.kill(); self.proc.wait(timeout=5)
        self.artifact.mkdir(parents=True, exist_ok=True)
        (self.artifact / "raw-events.json").write_text(json.dumps(self.raw, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def build_fixture(case: str) -> Path:
    """현재 정본을 SHA로 기록해 복사한다. legacy seed SHA와 동일하다고 가정하지 않는다."""
    root = ROOT / "runs" / case / "repo"
    if root.exists(): raise RuntimeError(f"existing artifact: {root}")
    root.mkdir(parents=True)
    seed = json.loads(SEED.read_text())
    for rel, body in seed.items():
        if rel in {"WORKFLOW_ASSET_SHA256.txt", ".gitignore"}: continue
        path = root / rel; path.parent.mkdir(parents=True, exist_ok=True); path.write_text(body, encoding="utf-8")
    for rel in ASSET_PATHS:
        src, dst = REPO / rel, root / rel
        dst.parent.mkdir(parents=True, exist_ok=True); shutil.copy2(src, dst)
    (root / ".agents/skills").mkdir(parents=True, exist_ok=True)
    os.symlink("../../.claude/skills/dev-cycle", root / ".agents/skills/dev-cycle")
    (root / "current-asset-sha.json").write_text(json.dumps({rel: sha(root / rel) for rel in ASSET_PATHS}, indent=2) + "\n")
    subprocess.run(["git", "init", "-b", "develop"], cwd=root, check=True, capture_output=True)
    subprocess.run(["git", "config", "user.name", "R2 Fixture"], cwd=root, check=True)
    subprocess.run(["git", "config", "user.email", "r2@example.invalid"], cwd=root, check=True)
    subprocess.run(["git", "add", "."], cwd=root, check=True); subprocess.run(["git", "commit", "-m", "test: r2 fixture"], cwd=root, check=True, capture_output=True)
    return root


PROMPTS = {"r2": "$dev-cycle next", "r3": "agent_type: dev-workflow 전용 역할로 인프라를 감사하고 결과와 TODO 초안은 대화로만 반환해 주세요.", "r4": "$dev-cycle status", "r5": "$dev-cycle next", "r7": "$dev-cycle status"}


def run_case(case: str) -> dict[str, Any]:
    repo = build_fixture(case); artifact = repo.parent
    rpc = Rpc(repo, artifact); result: dict[str, Any] = {"case": case, "actor": ACTOR, "repo": str(repo)}
    try:
        rpc.start(); root_id = rpc.thread("read-only" if case == "r3" else "workspace-write")
        result["root_thread"] = root_id
        result["events"] = rpc.turn(root_id, PROMPTS[case], skill=case != "r3")
        if case == "r3":
            loaded = rpc.call("thread/loaded/list", {}, 30)
            ids = extract_ids(loaded)
            result["loaded_thread_ids"] = sorted(ids)
            result["loaded_unrelated_ids"] = sorted(ids - {root_id})
            result["child_reads"] = {thread_id: rpc.call("thread/read", {"threadId": thread_id, "includeTurns": False}, 30) for thread_id in ids - {root_id}}
            child_records = [read.get("thread", read) for read in result["child_reads"].values()]
            result["r3_strict_role_evidence"] = [
                {"agentRole": row.get("agentRole"), "parentThreadId": row.get("parentThreadId"), "source": row.get("source"), "ephemeral": row.get("ephemeral")}
                for row in child_records
            ]
    finally: rpc.close()
    (artifact / "result.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    return result


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--case", choices=["r2", "r3", "r4", "r5", "r7"])
    parser.add_argument("--run", action="store_true", help="명시할 때만 실제 모델 turn을 실행")
    args = parser.parse_args()
    if args.run:
        if not args.case: parser.error("--run에는 --case가 필요합니다")
        result = {"execution": run_case(args.case)}
    else:
        result = {"local_regressions": local_checks(), "rpc_preparation": rpc_prepare(), "case": args.case or "all-preparation"}
    (ROOT / "preparation-result.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(result, ensure_ascii=False, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
