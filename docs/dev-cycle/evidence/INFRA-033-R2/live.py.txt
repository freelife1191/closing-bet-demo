def save(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def shell(repo, args, timeout=20):
    p = subprocess.run(args, cwd=repo, text=True, capture_output=True, timeout=timeout,
                       env={**filtered_env()[0], 'PYTHONDONTWRITEBYTECODE': '1'})
    return {'args': args, 'exit': p.returncode, 'stdout': p.stdout, 'stderr': p.stderr}


def git(repo, *args):
    r = shell(repo, ['git', *args])
    if r['exit']: raise RuntimeError(r)
    return r['stdout'].strip()


def snapshot(repo):
    tree, hooks = {}, {}
    for p in sorted(repo.rglob('*')):
        rel = str(p.relative_to(repo))
        if '.git' in p.parts or '__pycache__' in p.parts or not p.is_file(): continue
        (hooks if rel.startswith('.omx/') else tree)[rel] = hashlib.sha256(p.read_bytes()).hexdigest()
    return {'head': git(repo, 'rev-parse', 'HEAD'), 'status': git(repo, 'status', '--porcelain'),
            'tree': tree, 'hook_owned_observation': hooks}

class RPC:
    def __init__(self, repo):
        self.repo = repo; self.out = repo.parent; self.seq = 0
        self.events = queue.Queue(); self.raw = []; self.turns = []; self.denied = []
        self.proc = None; self.err = None

    def record(self, direction, msg):
        self.raw.append({'at': time.time(), 'direction': direction, 'message': msg})

    def start(self):
        env, removed = filtered_env()
        env['PYTHONDONTWRITEBYTECODE'] = '1'
        self.err = (self.out / 'stderr.log').open('w')
        trust = 'projects.' + json.dumps(str(self.repo)) + '.trust_level="trusted"'
        self.proc = subprocess.Popen([str(CODEX), '-c', trust, 'app-server', '--stdio'],
             cwd=self.repo, env=env, stdin=subprocess.PIPE, stdout=subprocess.PIPE,
             stderr=self.err, text=True, bufsize=1)
        save(self.out / 'launch.json', {'pid': self.proc.pid, 'argv': self.proc.args,
             'removed_env_keys': removed, 'model': 'gpt-5.6-sol', 'effort': 'high'})
        def reader():
            for line in self.proc.stdout:
                try: msg = json.loads(line)
                except json.JSONDecodeError: self.record('non-json', line); continue
                self.record('server', msg); self.events.put(msg)
        self.reader = threading.Thread(target=reader, daemon=True); self.reader.start()
        self.call('initialize', {'clientInfo': {'name': 'infra-033-r2', 'version': '2'},
                                'capabilities': {'experimentalApi': True}})
        self.send({'method': 'initialized', 'params': {}})

    def send(self, msg):
        self.record('client', msg)
        self.proc.stdin.write(json.dumps(msg, ensure_ascii=False) + '\n'); self.proc.stdin.flush()

    def receive(self, deadline):
        while time.monotonic() < deadline:
            try: msg = self.events.get(timeout=min(1, max(.01, deadline-time.monotonic())))
            except queue.Empty: continue
            if 'method' in msg and 'id' in msg:
                self.denied.append(msg['method'])
                self.send({'id': msg['id'], 'error': {'code': -32001, 'message': 'No interactive approvals in isolated evaluator'}})
                continue
            return msg
        raise TimeoutError('RPC wait')

    def call(self, method, params, timeout=30, error=False):
        self.seq += 1; ident = self.seq
        self.send({'id': ident, 'method': method, 'params': params})
        end = time.monotonic() + timeout
        while True:
            msg = self.receive(end)
            if msg.get('id') != ident: continue
            if 'error' in msg:
                if error: return msg
                raise RuntimeError(msg['error'])
            return msg.get('result', {})

    def thread(self, readonly=False):
        r = self.call('thread/start', {'cwd': str(self.repo), 'ephemeral': True,
             'sandbox': 'read-only' if readonly else 'workspace-write', 'approvalPolicy': 'never',
             'model': 'gpt-5.6-sol'})
        return r['thread']['id']

    def turn(self, thread, label, prompt, timeout=180, skill='dev-cycle'):
        inputs = []
        if skill:
            path = self.repo / '.agents/skills/dev-cycle/SKILL.md' if skill == 'dev-cycle' else Path('/Users/freelife/.codex/skills/ultraqa/SKILL.md')
            inputs.append({'type': 'skill', 'name': skill, 'path': str(path)})
        inputs.append({'type': 'text', 'text': prompt, 'text_elements': []})
        save(self.out / (label + '-input.json'), inputs)
        begin = len(self.raw)
        r = self.call('turn/start', {'threadId': thread, 'input': inputs, 'effort': 'high', 'approvalPolicy': 'never'})
        tid = r['turn']['id']; end = time.monotonic()+timeout
        result = {'threadId': thread, 'turnId': tid, 'label': label, 'timeoutSeconds': timeout, 'status': None}
        print(json.dumps({'case': self.repo.parent.name, 'turnStarted': label, 'thread': thread}), flush=True)
        try:
            while True:
                m = self.receive(end); p = m.get('params', {})
                if m.get('method') == 'item/completed' and p.get('item', {}).get('type') == 'agentMessage':
                    print(json.dumps({'case': self.repo.parent.name, 'message': p['item'].get('text','')[:300]}, ensure_ascii=False), flush=True)
                if m.get('method') == 'turn/completed' and p.get('threadId') == thread and p.get('turn', {}).get('id') == tid:
                    result['status'] = p['turn']['status']; break
        except TimeoutError:
            result['status'] = 'timeout'
            try: self.call('turn/interrupt', {'threadId': thread, 'turnId': tid}, timeout=15)
            except Exception as exc: result['interrupt_error'] = str(exc)
        result['events'] = self.raw[begin:]
        self.turns.append(result); save(self.out / (label + '-turn.json'), result)
        return result

    def close(self):
        if self.proc and self.proc.poll() is None:
            self.proc.terminate()
            try: self.proc.wait(timeout=10)
            except subprocess.TimeoutExpired: self.proc.kill(); self.proc.wait(timeout=5)
        if self.err: self.err.close()
        if hasattr(self, 'reader'): self.reader.join(timeout=2)
        save(self.out / 'raw-events.json', self.raw)
        save(self.out / 'cleanup.json', {'pid': self.proc.pid if self.proc else None,
             'returncode': self.proc.returncode if self.proc else None, 'denied_requests': self.denied})


def messages(raw, thread=None, final=False):
    found = []
    for event in raw:
        m=event.get('message', {}); p=m.get('params', {}); i=p.get('item', {})
        if m.get('method')=='item/completed' and i.get('type')=='agentMessage':
            if thread and p.get('threadId')!=thread: continue
            if final and i.get('phase')!='final_answer': continue
            found.append(i.get('text',''))
    return found
