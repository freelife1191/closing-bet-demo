import requests
from urllib.parse import urlsplit
_original = requests.Session.request
def request(self, method, url, **kwargs):
    if urlsplit(str(url)).hostname == 'data.krx.co.kr':
        response = requests.Response()
        response.status_code = 200
        response.url = str(url)
        response._content = b'<html>synthetic KRX maintenance</html>'
        response.headers['Content-Type'] = 'text/html'
        return response
    return _original(self, method, url, **kwargs)
requests.Session.request = request

import socket
_connect = socket.socket.connect
_connect_ex = socket.socket.connect_ex
def local_only(address):
    if isinstance(address, tuple) and address[0] not in ('127.0.0.1', 'localhost', '::1'):
        raise OSError('QA blocked external network')
def connect(self, address):
    local_only(address)
    return _connect(self, address)
def connect_ex(self, address):
    local_only(address)
    return _connect_ex(self, address)
socket.socket.connect = connect
socket.socket.connect_ex = connect_ex
